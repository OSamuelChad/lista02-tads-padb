AS $$
BEGIN
    -- "Verifique a existência": se o cliente ainda não tem linha de
    -- estatística (ex.: cliente novo), cria com 1; senão, incrementa.
    -- O upsert é atômico e evita condição de corrida entre UPDATE e INSERT.
    INSERT INTO customer_rental_stats (customer_id, total_rentals)
    VALUES (NEW.customer_id, 1)
    ON CONFLICT (customer_id)
    DO UPDATE SET total_rentals = customer_rental_stats.total_rentals + 1;

    RETURN NULL;
END;
$$;
